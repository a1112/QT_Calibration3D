cmake_minimum_required(VERSION 3.16)
project(FileContentTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core)
add_executable(filecontent_test filecontent_test.cpp ../filecontent.cpp ../filecontent.h)
target_include_directories(filecontent_test PRIVATE ..)
target_link_libraries(filecontent_test PRIVATE Qt6::Core)
enable_testing()
add_test(NAME filecontent_recovery COMMAND filecontent_test)
